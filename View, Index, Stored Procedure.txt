CREATE DATABASE IF NOT EXISTS ProductDemo;
USE ProductDemo;

CREATE TABLE IF NOT EXISTS Products (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    productCode VARCHAR(20) NOT NULL,
    productName VARCHAR(100) NOT NULL,
    productPrice DECIMAL(12,2) NOT NULL,
    productAmount INT NOT NULL DEFAULT 0,
    productDescription TEXT,
    productStatus TINYINT NOT NULL DEFAULT 1
);

CREATE OR REPLACE VIEW product_views AS
SELECT productCode, productName,
       productPrice, productStatus
FROM Products;

SELECT * FROM product_views;

-- Sửa View: chỉ lấy sản phẩm đang hoạt động
CREATE OR REPLACE VIEW product_views AS
SELECT productCode, productName,
       productPrice, productStatus
FROM Products
WHERE productStatus = 1;

SELECT * FROM product_views;

-- Xóa View
DROP VIEW IF EXISTS product_views;

-- Bước 5: Stored Procedure lấy tất cả sản phẩm
DROP PROCEDURE IF EXISTS sp_getAllProducts;

DELIMITER //

CREATE PROCEDURE sp_getAllProducts()
BEGIN
    SELECT * FROM Products;
END //

DELIMITER ;

-- Stored Procedure thêm sản phẩm
DROP PROCEDURE IF EXISTS sp_addProduct;

DELIMITER //

CREATE PROCEDURE sp_addProduct(
    IN p_code VARCHAR(20),
    IN p_name VARCHAR(100),
    IN p_price DECIMAL(12,2),
    IN p_amount INT,
    IN p_description TEXT,
    IN p_status TINYINT
)
BEGIN
    INSERT INTO Products
    (productCode, productName, productPrice,
     productAmount, productDescription, productStatus)
    VALUES
    (p_code, p_name, p_price, p_amount,
     p_description, p_status);
END //

DELIMITER ;

-- Stored Procedure sửa sản phẩm theo Id
DROP PROCEDURE IF EXISTS sp_updateProduct;

DELIMITER //

CREATE PROCEDURE sp_updateProduct(
    IN p_id INT,
    IN p_code VARCHAR(20),
    IN p_name VARCHAR(100),
    IN p_price DECIMAL(12,2),
    IN p_amount INT,
    IN p_description TEXT,
    IN p_status TINYINT
)
BEGIN
    UPDATE Products
    SET productCode = p_code,
        productName = p_name,
        productPrice = p_price,
        productAmount = p_amount,
        productDescription = p_description,
        productStatus = p_status
    WHERE Id = p_id;
END //

DELIMITER ;

-- Stored Procedure xóa sản phẩm theo Id
DROP PROCEDURE IF EXISTS sp_deleteProduct;

DELIMITER //

CREATE PROCEDURE sp_deleteProduct(IN p_id INT)
BEGIN
    DELETE FROM Products
    WHERE Id = p_id;
END //

DELIMITER ;

-- Kiểm tra các Stored Procedure
CALL sp_getAllProducts();

CALL sp_addProduct(
    'P006', 'Tablet Samsung', 5990000, 5,
    'May tinh bang', 1
);

CALL sp_getAllProducts();

CALL sp_updateProduct(
    6, 'P006', 'Tablet Samsung Updated',
    5790000, 7, 'May tinh bang da cap nhat', 1
);

CALL sp_getAllProducts();

CALL sp_deleteProduct(6);

CALL sp_getAllProducts();
